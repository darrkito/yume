# One group for everyone: a crawler obeys only the most specific group that
# names it, so separate per-bot "Allow: /" groups silently dropped these
# Disallows for Googlebot, Bingbot and the AI bots. The named tokens below
# change nothing a bare "*" would not already allow; they document that each
# kind of AI crawler (training, search, user-triggered fetch) is welcome.
#
# Cart/checkout pages are NOT disallowed on purpose: they carry
# <meta name="robots" content="noindex">, and a Disallow would stop crawlers
# from ever seeing it (a blocked URL can still be indexed from links alone).
User-agent: *
User-agent: Googlebot
User-agent: Bingbot
User-agent: Applebot
# Training crawlers
User-agent: GPTBot
User-agent: ClaudeBot
User-agent: Google-Extended
User-agent: Applebot-Extended
User-agent: Meta-ExternalAgent
User-agent: CCBot
User-agent: Amazonbot
User-agent: Bytespider
# AI search crawlers
User-agent: OAI-SearchBot
User-agent: Claude-SearchBot
User-agent: PerplexityBot
User-agent: DuckAssistBot
# User-triggered fetchers
User-agent: ChatGPT-User
User-agent: Claude-User
User-agent: Perplexity-User
User-agent: MistralAI-User

Disallow: /api/
